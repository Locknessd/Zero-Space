using UnityEngine;

namespace MemeArena
{
    public sealed class MemeCrowdFan : MonoBehaviour
    {
        [Min(0)] public float bounceHeight = 0.12f;
        [Min(0.1f)] public float speed = 2.5f;
        [Range(0, 30)] public float swayDegrees = 7f;
        private Vector3 origin;
        private Quaternion rotation;
        private float phase;
        private float boost = 1f;
        private float boostEnd;

        private void Awake()
        {
            origin = transform.localPosition;
            rotation = transform.localRotation;
            phase = Random.Range(0f, 6.28318f);
        }

        public void Cheer(float intensity = 2f, float duration = 1.5f)
        {
            boost = Mathf.Max(1f, intensity);
            boostEnd = Time.time + Mathf.Max(0f, duration);
        }

        private void Update()
        {
            float multiplier = Time.time < boostEnd ? boost : 1f;
            float t = Time.time * speed + phase;
            transform.localPosition = origin + Vector3.up * (Mathf.Abs(Mathf.Sin(t)) * bounceHeight * multiplier);
            transform.localRotation = rotation * Quaternion.Euler(0, 0, Mathf.Sin(t * 0.7f) * swayDegrees * multiplier);
        }
    }
}
